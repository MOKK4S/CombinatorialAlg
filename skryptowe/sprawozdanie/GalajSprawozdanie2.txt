#!/bin/bash
file="$1"

prev_resid=""
prev_chain=""

while IFS= read -r line
do
    if [[ $line == ATOM* ]]; then
        resname=$(echo "$line" | awk '{print $4}')
        resid=$(echo "$line" | awk '{print $6}')
        chain=$(echo "$line" | awk '{print $5}')

            echo -n "$resname "
        fi

        prev_resid=$resid
        prev_chain=$chain
    fi
done < "$file"

echo

#!/bin/bash
file=$1

one_letter() {
    case "$1" in
        ALA) echo "A" ;;
        ARG) echo "R" ;;
        ASN) echo "N" ;;
        ASP) echo "D" ;;
        CYS) echo "C" ;;
        GLN) echo "Q" ;;
        GLU) echo "E" ;;
        GLY) echo "G" ;;
        HIS) echo "H" ;;
        ILE) echo "I" ;;
        LEU) echo "L" ;;
        LYS) echo "K" ;;
        MET) echo "M" ;;
        PHE) echo "F" ;;
        PRO) echo "P" ;;
        SER) echo "S" ;;
        THR) echo "T" ;;
        TRP) echo "W" ;;
        TYR) echo "Y" ;;
        VAL) echo "V" ;;
        *) echo "X" ;;
    esac
}

prev_res=""
while IFS= read -r line; do
    if [[ ${line:0:4} == "ATOM" ]]; then

        res=$(echo "$line" | cut -c 18-20)

        id=$(echo "$line" | cut -c 23-26 | tr -d " ")

        key="$res-$id"

        if [[ "$key" != "$prev_res" ]]; then
            echo -n "$(one_letter "$res")"
            prev_res="$key"
        fi
    fi
done < "$file"

echo
